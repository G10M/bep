require('dotenv').config();
const express = require('express');
const puppeteer = require('puppeteer');
const rateLimit = require('express-rate-limit');
const basicAuth = require('express-basic-auth');

const app = express();

// 1. Защита паролем (чтобы друг зашел по логину/паролю)
app.use(basicAuth({
    users: { [process.env.APP_USER || 'friend']: process.env.APP_PASS || '12345' },
    challenge: true,
    realm: 'Rubep Balance'
}));

// Раздаем файлы из папки public
app.use(express.static('public'));

// 2. Защита от спама (макс 5 запросов в минуту)
const limiter = rateLimit({ windowMs: 60_000, max: 5 });
app.use('/api/', limiter);

// 3. Кэш, чтобы не запускать браузер каждую секунду
let cache = { value: null, ts: 0 };
const CACHE_TTL = 60_000; // Данные живут 1 минуту

async function fetchBalanceFromRubep() {
  // Запускаем невидимый браузер. На хостинге нужны флаги безопасности.
  const browser = await puppeteer.launch({
    headless: 'new',
    args: ['--no-sandbox', '--disable-setuid-sandbox', '--disable-dev-shm-usage']
  });

  try {
    const page = await browser.newPage();
    await page.setUserAgent(
      'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 ' +
      '(KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36'
    );

    console.log('Открываем страницу логина...');
    await page.goto('https://app.rubep.com/', {
      waitUntil: 'networkidle2',
      timeout: 60_000
    });

    console.log('Вводим данные...');
    await page.waitForSelector('input[type="email"]', { timeout: 30_000 });
    await page.type('input[type="email"]', process.env.RUBEP_LOGIN);
    await page.type('input[type="password"]', process.env.RUBEP_PASSWORD);

    console.log('Ждем ответ от API при входе...');
    
    // Перехватываем JSON-ответ
    const responsePromise = page.waitForResponse(async response => {
      if (response.url().includes('/api/') && response.request().method() === 'GET') {
        try {
          const json = await response.json();
          if (json && json.success && json.data && json.data.cards) {
            return true;
          }
        } catch (e) {
          return false;
        }
      }
      return false;
    }, { timeout: 30_000 });

    await page.click('button[type="submit"]');

    const response = await responsePromise;
    const json = await response.json();

    const cards = json.data.cards;
    if (!cards || cards.length === 0) {
      throw new Error('В ответе API нет карт');
    }

    const balance = cards[0].balance;
    console.log(`Успешно получен баланс: ${balance}`);
    
    return balance;

  } finally {
    await browser.close();
  }
}

// 4. Наш API для фронтенда
app.get('/api/balance', async (req, res) => {
  try {
    const now = Date.now();
    if (cache.value && now - cache.ts < CACHE_TTL) {
      return res.json({ balance: cache.value, cached: true });
    }

    const balance = await fetchBalanceFromRubep();
    cache = { value: balance, ts: now };
    
    res.json({ balance, cached: false });
  } catch (err) {
    console.error('Ошибка получения баланса:', err.message);
    res.status(502).json({ error: 'Не удалось получить баланс' });
  }
});

// 5. Запуск сервера (0.0.0.0 нужен для Render)
const PORT = process.env.PORT || 3000;
app.listen(PORT, '0.0.0.0', () => {
  console.log(`Сервер запущен на порту ${PORT}`);
});
